python — export to csv
import requests, csv

BASE = "https://publicdrop.in/APIv3"
coins = []

for page in range(1, 3):
    r = requests.get(f"{BASE}/coins", params={"page":page,"sort":"rank"})
    coins.extend(r.json()["coins"])

with open('crypto.csv', 'w', newline='') as f:
    w = csv.DictWriter(f, fieldnames=['rank','name','symbol','price_usd'])
    w.writeheader()
    for c in coins:
        w.writerow({'rank':c['rank'],'name':c['name'],'symbol':c['symbol'],'price_usd':c.get('price_usd')})
print(f"Exported {len(coins)} coins")
